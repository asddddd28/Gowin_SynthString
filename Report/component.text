\begin{equation}
  R = \frac{N}{t}

\end{equation}
式~\eqref{eq:component-throughput} 中，\(N\) 为处理的数据量，\(t\) 为耗时。

% ==================== 09 多行对齐公式 ====================
% align 中每行独立编号，& 标记对齐位置。
\begin{align}
  T_{\mathrm{clk}} &= \frac{1}{f_{\mathrm{clk}}},
  \label{eq:component-clock}\\
  t_{\mathrm{latency}} &= L\,T_{\mathrm{clk}}.
  \label{eq:component-latency}
\end{align}

  \begin{aligned}
    y[n] &= \sum_{k=0}^{M-1}h[k]x[n-k],\\
    H(z) &= \sum_{k=0}^{M-1}h[k]z^{-k}.
  \end{aligned}
  \label{eq:component-filter}
\end{equation}

  y = \begin{cases}

  \end{cases}
  \label{eq:component-piecewise}
\end{equation}

% ==================== 11 HDL 代码块 ====================
% caption 自动编号；label 支持交叉引用。代码保留原文及缩进。
\begin{lstlisting}[language=Verilog,caption={流水寄存器示例},label={lst:component-register}]
always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
        data_out <= 16'd0;
    else if (valid_in)
        data_out <= data_in;
end
\end{lstlisting}
寄存器更新过程见代码~\ref{lst:component-register}。
